const REPLY =
'Hi! I can help you book an appointment, answer questions, or transfer you to a person. Try cutting me off mid-sent—';

const WAVE: Record<Phase, WaveState> = {
  idle: 'idle',
  listening: 'listening',
  thinking: 'thinking',
  speaking: 'speaking',
  interrupted: 'idle',
  done: 'idle'
};

const STATUS: Record<Phase, string> = {
  idle: 'Click the button and try interrupting me mid-sentence',
  listening: 'Listening — it picks up your voice as you speak',
  thinking: 'Thinking — working out what to say back',
  speaking: 'Speaking — go ahead, cut it off',
  interrupted: '',
  done: 'It finished on its own that time. Try again and interrupt it.'
};

export function InteractiveDemo() {
  const [phase, setPhase] = useState<Phase>('idle');
  const [typed, setTyped] = useState(0);

  useEffect(() => {
    if (phase === 'listening') {
      const t = window.setTimeout(() => setPhase('thinking'), 2000);
      return () => window.clearTimeout(t);
    }

    if (phase === 'thinking') {
      const t = window.setTimeout(() => setPhase('speaking'), 800);
      return () => window.clearTimeout(t);
    }

    if (phase === 'speaking') {
      let i = 0;
      setTyped(0);
      const id = window.setInterval(() => {
        i += 1;
        setTyped(i);
        if (i >= REPLY.length) {
          window.clearInterval(id);
          setPhase('done');
        }
      }, 30);
      return () => window.clearInterval(id);
    }

    return undefined;
  }, [phase]);

  const start = () => {
    setTyped(0);
    setPhase('listening');
  };

  const interrupt = () => setPhase('interrupted');

  const reset = () => {
    setTyped(0);
    setPhase('idle');
  };

  const busy = phase === 'listening' || phase === 'thinking' || phase === 'speaking';
  const showBubble = phase === 'speaking' || phase === 'interrupted' || phase === 'done';
  const buttonLabel =
  phase === 'listening' ?
  'Listening...' :
  phase === 'thinking' ?
  'Thinking...' :
  phase === 'speaking' ?
  'Speaking...' :
  'Tap to talk';

  return (
    <motion.div data-ev-id="ev_a3d9e4e376"
    className="mx-auto w-full max-w-[600px] rounded-[32px] p-8 lg:p-10"
    style={{
      background: 'var(--theme-surface)',
      boxShadow: 'var(--theme-shadow-lg)',
      border: '1px solid var(--theme-border)'

    whileInView={{ opacity: 1, y: 0 }}
    viewport={{ once: true }}
    transition={{ duration: 0.6, ease: 'easeOut' }}>

					</AnimatePresence>
				</div>

function usePrefersReducedMotion() {
  const [reduced, setReduced] = useState(false);

  useEffect(() => {
    const mq = window.matchMedia('(prefers-reduced-motion: reduce)');
    setReduced(mq.matches);
    const onChange = (e: MediaQueryListEvent) => setReduced(e.matches);
    mq.addEventListener('change', onChange);
    return () => mq.removeEventListener('change', onChange);
  }, []);

  return reduced;
}

  idle: 'Waveform idle',
  listening: 'Waveform listening',
  thinking: 'Waveform thinking',
  speaking: 'Waveform speaking'

export function Waveform({
  state,
  height = 80,
  barCount = 9,
  barWidth = 6,
  gap = 6,
  className = '',
  label
}: WaveformProps) {
  const { theme } = useTheme();
  const reduced = usePrefersReducedMotion();
  const [levels, setLevels] = useState<number[]>(() => Array(barCount).fill(0.4));
  const timerRef = useRef<number | null>(null);

  const accentColor = theme === 'dark' ? '#FF7A1A' : '#FF6B00';
  const accentHover = theme === 'dark' ? '#FF8F3D' : '#FF8534';
  const successColor = theme === 'dark' ? '#2DD4BF' : '#10B981';

  useEffect(() => {
    if (state !== 'listening' || reduced) {
      if (timerRef.current) {
        window.clearInterval(timerRef.current);
        timerRef.current = null;
      }
      return;
    }

    const tick = () =>
    setLevels(Array.from({ length: barCount }, () => 0.2 + Math.random() * 0.7));

    tick();
    timerRef.current = window.setInterval(tick, 150);

    return () => {
      if (timerRef.current) {
        window.clearInterval(timerRef.current);
        timerRef.current = null;
      }
    };
  }, [state, barCount, reduced]);

  const bars = useMemo(() => Array.from({ length: barCount }, (_, i) => i), [barCount]);

  const isThinking = state === 'thinking';
  const dotSize = Math.max(6, Math.round(height * 0.15));
  const width = barCount * barWidth + (barCount - 1) * gap;

  return (
    <div data-ev-id="ev_1fa8bd9d9d"
    className={`relative flex items-center justify-center ${className}`}
    style={{ height, width }}

			<div data-ev-id="ev_be95981d11"
      className="flex items-center justify-center transition-opacity duration-300"
      style={{ gap, height, opacity: isThinking ? 0 : 1 }}
      aria-hidden="true">

				{bars.map((i) => {
          const style: React.CSSProperties = {
            width: barWidth,
            height: '100%',
            borderRadius: Math.min(4, barWidth / 2),
            transformOrigin: 'center',
            backgroundColor: state === 'listening' ? accentHover : accentColor,
            transition: 'background-color 300ms ease-out, transform 150ms ease-out'
          };

          if (reduced) {
            style.transform = `scaleY(${state === 'idle' ? 0.2 : 0.6})`;
          } else if (state === 'listening') {
            style.transform = `scaleY(${levels[i] ?? 0.3})`;
          } else if (state === 'speaking') {
            style.animation = `speaking-wave 1.2s ease-in-out ${(-(i * 1.2) / barCount).toFixed(3)}s infinite`;
          } else {
            style.animation = `idle-breathe 3s ease-in-out ${(i * 0.1).toFixed(2)}s infinite`;

const STORAGE_KEY = 'voiceai-theme';

export function ThemeProvider({ children }: { children: ReactNode }) {
	const [theme, setThemeState] = useState<Theme>('light');
	const [mounted, setMounted] = useState(false);

	useEffect(() => {
		const stored = localStorage.getItem(STORAGE_KEY) as Theme | null;
		const prefersDark = window.matchMedia('(prefers-color-scheme: dark)').matches;
		const initial = stored ?? (prefersDark ? 'dark' : 'light');
		setThemeState(initial);
		document.documentElement.setAttribute('data-theme', initial);
		setMounted(true);
	}, []);

	useEffect(() => {
		if (!mounted) return;
		document.documentElement.setAttribute('data-theme', theme);
		localStorage.setItem(STORAGE_KEY, theme);
	}, [theme, mounted]);

	const toggleTheme = () => setThemeState((t) => (t === 'light' ? 'dark' : 'light'));
	const setTheme = (t: Theme) => setThemeState(t);

	return (
		<ThemeContext.Provider value={{ theme, toggleTheme, setTheme }}>
			{children}
		</ThemeContext.Provider>
	);
}
